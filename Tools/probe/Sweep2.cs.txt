// A sweep of every country in the world: stand in each in turn and report what the ground under
// you actually is, how much is standing on it, and what a frame costs. One run, one table, so a
// change that quietly wrecks one country shows up without anybody having to go and look at all
// fourteen.
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        AudioListener.volume = 0f; AudioListener.pause = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    /// <summary>Whether a tile definition is the country's own ground rather than a neighbour's.</summary>
    private static bool OwnGround(Regions.Character who, int id)
    {
        switch (who)
        {
            case Regions.Character.Forest: return id >= 35 && id <= 39;
            case Regions.Character.Lowland: case Regions.Character.Hills: return id >= 40 && id <= 54;
            case Regions.Character.Reed: return id >= 55 && id <= 59;
            case Regions.Character.Water: return id >= 60 && id <= 64 || id >= 110 && id <= 114;
            case Regions.Character.Desert: return id >= 65 && id <= 69;
            case Regions.Character.Stone: return id >= 70 && id <= 79;
            case Regions.Character.Snow: return id >= 80 && id <= 84;
            case Regions.Character.Fungal: return id >= 85 && id <= 89;
            case Regions.Character.Dead: return id >= 90 && id <= 94;
            case Regions.Character.Reef: return id >= 95 && id <= 99 || id >= 195 && id <= 199 || id >= 60 && id <= 64;
            case Regions.Character.Jungle: return id >= 100 && id <= 104;
            case Regions.Character.Peaks: return id >= 105 && id <= 109 || id >= 75 && id <= 84;
            case Regions.Character.Savanna: return id >= 165 && id <= 169;
            default: return false;
        }
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        var save = WorldLibrary.Create("Probe sweep", 5, true, true, 0.42f, 20f, true, true);
        WorldLibrary.Enter(save);
        yield return new WaitForSecondsRealtime(9f);
        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var arrival = FindFirstObjectByType<Arrival>();
        if (arrival != null) typeof(Arrival).GetMethod("Close", flags).Invoke(arrival, null);
        TimeOfDay.Instance.SetTime(0.42f); TimeOfDay.Instance.Paused = true;

        var world = FindFirstObjectByType<ChunkManager>();
        var player = world.PlayerTransform;
        int seed = world.WorldSeed;
        var chunks = (Dictionary<Vector2Int, Chunk>)typeof(ChunkManager).GetField("chunks", flags).GetValue(world);
        var body = FindFirstObjectByType<CharacterController>();
        var under = FindFirstObjectByType<Undergrowth>();

        Mark("SWEEP country          | own ground | mixed | standing | frame (median of four seconds)");

        foreach (Regions.Character want in System.Enum.GetValues(typeof(Regions.Character)))
        {
            int standX = 0, standZ = 0; bool found = false;
            for (int r = 1; r < 500 && !found; r++)
            for (int ox = -r; ox <= r && !found; ox++)
            for (int oz = -r; oz <= r && !found; oz++)
            {
                if (Mathf.Max(Mathf.Abs(ox), Mathf.Abs(oz)) != r) continue;
                if (Regions.CharacterAtTile(ox, oz, seed, false) != want) continue;
                if (WaterSurface.IsUnderwater(ox, oz, seed)) continue;
                bool inside = true;
                for (int dx = -30; dx <= 30 && inside; dx += 10)
                for (int dz = -30; dz <= 30 && inside; dz += 10)
                    if (Regions.CharacterAtTile(ox + dx, oz + dz, seed, false) != want) inside = false;
                if (!inside) continue;
                standX = ox; standZ = oz; found = true;
            }
            if (!found) { Mark(string.Format("  {0,-16} none found within 500 tiles", want)); continue; }

            if (body != null) body.enabled = false;
            player.position = new Vector3(standX * WorldGrid.TileSize, WorldHeight.SurfaceY(standX, standZ, seed) + 1.4f, standZ * WorldGrid.TileSize);
            if (body != null) body.enabled = true;
            yield return new WaitForSecondsRealtime(10f);

            var centre = WorldGrid.WorldToChunk(player.position);
            var ids = new SortedDictionary<int, int>();
            int drawn = 0;
            for (int dx = -1; dx <= 1; dx++)
            for (int dz = -1; dz <= 1; dz++)
            {
                if (!chunks.TryGetValue(centre + new Vector2Int(dx, dz), out var chunk)) continue;
                foreach (var pair in chunk.idToTransforms)
                {
                    drawn += pair.Value.Length;
                    ids[pair.Key] = (ids.TryGetValue(pair.Key, out int n) ? n : 0) + pair.Value.Length;
                }
            }
            int most = 0, mostId = -1, blended = 0;
            foreach (var pair in ids)
            {
                if (pair.Value > most) { most = pair.Value; mostId = pair.Key; }
                // 115 to 194 less the savanna's own 165 to 169, which sit in the middle of
                // that run and were being counted as mixed ground, putting the plain at 94%
                if (pair.Key >= 115 && pair.Key <= 194 && (pair.Key < 165 || pair.Key > 169)) blended += pair.Value;
            }

            int standing = 0;
            var counts = under != null ? (int[])typeof(Undergrowth).GetField("counts", flags).GetValue(under) : null;
            if (counts != null) foreach (int n in counts) standing += n;

            // The median of a longer run rather than the mean of two seconds. A single hitch
            // in a short window once put the peaks at 9.0 ms against a true 3.2, which reads as
            // a threefold regression and costs a run to disprove.
            var spans = new List<float>();
            for (float t = 0f; t < 4f; t += Time.unscaledDeltaTime) { spans.Add(Time.unscaledDeltaTime); yield return null; }
            spans.Sort();
            float middle = spans.Count > 0 ? spans[spans.Count / 2] * 1000f : 0f;
            float worst = spans.Count > 0 ? spans[spans.Count - 1] * 1000f : 0f;

            // and what the country's own ground comes to, which is the number that matters and
            // is not the same as the commonest tile: the savanna's bug hid behind exactly that
            int own = 0;
            foreach (var pair in ids) if (OwnGround(want, pair.Key)) own += pair.Value;

            Mark(string.Format("  {0,-16} | own {1,3:F0}% | mixed {2,3:F0}% | {3,6} standing | {4,4:F1} ms (worst {5,4:F1})",
                want, 100f * own / Mathf.Max(1, drawn), 100f * blended / Mathf.Max(1, drawn),
                standing, middle, worst));
        }
        Mark("done");
        WorldLibrary.Delete(save.id);
    }
}
