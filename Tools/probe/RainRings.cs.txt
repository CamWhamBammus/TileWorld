// A probe: rain over a lake, photographed from the shore, with the rings counted.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        if (TitleMenu.IsUp)
        {
            var worlds = WorldLibrary.All();
            WorldLibrary.Enter(worlds.Count > 0 ? worlds[0] : WorldLibrary.Create("Rain", 7777));
            yield return new WaitForSecondsRealtime(6f);
        }
        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }
        var player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        yield return new WaitForSecondsRealtime(4f);
        HideInterface();

        // the nearest shore tile with water three tiles out in some direction
        int px = Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), pz = Mathf.RoundToInt(player.position.z / WorldGrid.TileSize);
        int bestX = 0, bestZ = 0, bestDx = 0, bestDz = 0; bool found = false;
        int[] dirs = { 1, 0, -1, 0, 0, 1, 0, -1 };
        for (int r = 4; r < 110 && !found; r += 2)
        for (int dx = -r; dx <= r && !found; dx += 2)
        for (int dz = -r; dz <= r && !found; dz += 2)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
            int tx = px + dx, tz = pz + dz;
            if (WaterSurface.IsUnderwater(tx, tz, seed)) continue;
            for (int d = 0; d < 4 && !found; d++)
            {
                int wet = 0;
                for (int k = 2; k <= 8; k++) if (WaterSurface.IsUnderwater(tx + dirs[d * 2] * k, tz + dirs[d * 2 + 1] * k, seed)) wet++;
                if (wet >= 6) { bestX = tx; bestZ = tz; bestDx = dirs[d * 2]; bestDz = dirs[d * 2 + 1]; found = true; }
            }
        }
        if (!found) { Mark("no shore"); Mark("done"); yield break; }

        Vector3 shore = new Vector3(bestX * WorldGrid.TileSize, WorldHeight.SurfaceY(bestX, bestZ, seed) + 1f, bestZ * WorldGrid.TileSize);
        Vector3 into = new Vector3(bestDx, 0f, bestDz);
        body.enabled = false; player.position = shore; player.rotation = Quaternion.LookRotation(into); body.enabled = true;
        yield return new WaitForSecondsRealtime(3f);

        var main = Camera.main;
        var cam = new GameObject("ProbeCam").AddComponent<Camera>();
        cam.CopyFrom(main); cam.depth = main.depth + 50f; cam.fieldOfView = 50f;
        cam.transform.position = shore + Vector3.up * 1.6f - into * 1.5f;
        cam.transform.LookAt(new Vector3(shore.x, WaterSurface.Level, shore.z) + into * 9f);

        Mark("dry: rain rings " + Splashes.RainAlive + ", overcast " + TimeOfDay.Instance.Overcast.ToString("F2"));
        TimeOfDay.Instance.ForceOvercast(1f);
        yield return new WaitForSecondsRealtime(4f);

        var rain = FindFirstObjectByType<Rain>();
        if (rain == null) Mark("RAIN COMPONENT: none in the scene");
        else
        {
            int active = 0, overWater = 0; float lo = 999f, hi = -999f;
            foreach (Transform t in rain.transform)
            {
                if (!t.gameObject.activeSelf) continue;
                active++;
                float y = t.position.y; lo = Mathf.Min(lo, y); hi = Mathf.Max(hi, y);
                if (WaterSurface.IsUnderwater(Mathf.RoundToInt(t.position.x / WorldGrid.TileSize), Mathf.RoundToInt(t.position.z / WorldGrid.TileSize), seed)) overWater++;
            }
            Mark("RAIN COMPONENT: enabled " + rain.enabled + ", at " + rain.transform.position.ToString("F1") + ", main camera at " + (Camera.main != null ? Camera.main.transform.position.ToString("F1") : "none")
                 + ", active streaks " + active + " (" + rain.transform.childCount + " made), over water " + overWater + ", streak y " + lo.ToString("F1") + " to " + hi.ToString("F1") + ", water level " + WaterSurface.Level.ToString("F1") + ", shore " + shore.y.ToString("F1"));
        }
        Vector3 spot = new Vector3(shore.x, WaterSurface.Level, shore.z) + into * 6f;
        Splashes.Raindrop(spot, seed);
        yield return null;
        Mark("BY HAND: a raindrop at " + spot.ToString("F1") + " -> rain rings " + Splashes.RainAlive + " (underwater there: " + WaterSurface.IsUnderwater(Mathf.RoundToInt(spot.x / WorldGrid.TileSize), Mathf.RoundToInt(spot.z / WorldGrid.TileSize), seed) + ")");
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-rr-rain.png"));
        Splashes.RainSpread(Camera.main.transform.position, out int near, out int mid, out int far);
        Mark("RAIN: rings on the water " + Splashes.RainAlive + " (within 15 m " + near + ", 15-40 m " + mid + ", beyond " + far + "), overcast " + TimeOfDay.Instance.Overcast.ToString("F2") + ", wind " + Rain.Wind.magnitude.ToString("F1") + " m/s, snowing " + Rain.Snowing + ", patter " + Splashes.Patter.ToString("F2"));

        // what it costs: frames over two seconds raining, then two seconds after
        int frames = 0; float t0 = Time.realtimeSinceStartup;
        while (Time.realtimeSinceStartup - t0 < 2f) { frames++; yield return null; }
        float raining = 2000f / frames;
        yield return new WaitForSecondsRealtime(1f);
        cam.transform.position = shore + Vector3.up * 6f - into * 4f;
        cam.transform.LookAt(new Vector3(shore.x, WaterSurface.Level, shore.z) + into * 30f);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-rr-rain2.png"));
        yield return new WaitForSecondsRealtime(0.5f);
        TimeOfDay.Instance.ForceOvercast(0f);
        yield return new WaitForSecondsRealtime(2f);
        frames = 0; t0 = Time.realtimeSinceStartup;
        while (Time.realtimeSinceStartup - t0 < 2f) { frames++; yield return null; }
        Mark("COST: " + raining.ToString("F2") + " ms a frame in a downpour, " + (2000f / frames).ToString("F2") + " ms clear");
        TimeOfDay.Instance.ForceOvercast(-1f);
        Mark("after: rain rings " + Splashes.RainAlive + ", patter " + Splashes.Patter.ToString("F2"));

        // the snow country: it snows, and the lakes are ice you stand on
        int fx = 0, fz = 0, dx2 = 0, dz2 = 0; bool frozenFound = false;
        for (int r = 6; r < 420 && !frozenFound; r += 3)
        for (int dx = -r; dx <= r && !frozenFound; dx += 3)
        for (int dz = -r; dz <= r && !frozenFound; dz += 3)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
            int tx = bestX + dx, tz = bestZ + dz;
            if (!WaterSurface.IsFrozen(tx, tz, seed)) continue;
            for (int d = 0; d < 4 && !frozenFound; d++)
            {
                int nx = tx + dirs[d * 2] * 3, nz = tz + dirs[d * 2 + 1] * 3;
                if (WaterSurface.IsUnderwater(nx, nz, seed)) continue;
                if (Regions.CharacterAtTile(nx, nz, seed, false) != Regions.Character.Snow) continue;
                fx = nx; fz = nz; dx2 = -dirs[d * 2]; dz2 = -dirs[d * 2 + 1]; frozenFound = true;
            }
        }
        if (!frozenFound) { Mark("SNOW: no frozen lake within 420 tiles of here"); Mark("done"); yield break; }

        Vector3 bank = new Vector3(fx * WorldGrid.TileSize, WorldHeight.SurfaceY(fx, fz, seed) + 1f, fz * WorldGrid.TileSize);
        Vector3 onto = new Vector3(dx2, 0f, dz2);
        body.enabled = false; player.position = bank; player.rotation = Quaternion.LookRotation(onto); body.enabled = true;
        yield return new WaitForSecondsRealtime(6f);
        TimeOfDay.Instance.ForceOvercast(1f);
        yield return new WaitForSecondsRealtime(4f);

        int iceTiles = 0;
        for (int dx = -10; dx <= 10; dx++) for (int dz = -10; dz <= 10; dz++) if (WaterSurface.IsFrozen(fx + dx, fz + dz, seed)) iceTiles++;
        Mark("SNOW: at " + bank.ToString("F0") + ", snowing " + Rain.Snowing + ", rain rings " + Splashes.RainAlive + ", patter " + Splashes.Patter.ToString("F2") + ", ice tiles within 10: " + iceTiles + ", ice objects " + GameObject.Find("Ice (runtime)").transform.childCount);

        // what the ice actually is here: the patch, its mesh, its material, and what a ray finds
        var chunkAt = WorldGrid.WorldToChunk(bank + onto * 6f);
        var icePatch = GameObject.Find("Ice " + chunkAt); var waterPatch = GameObject.Find("Water " + chunkAt);
        string iceSay = icePatch == null ? "no ice patch" : "ice patch verts " + (icePatch.GetComponent<MeshFilter>() != null && icePatch.GetComponent<MeshFilter>().sharedMesh != null ? icePatch.GetComponent<MeshFilter>().sharedMesh.vertexCount.ToString() : "none");
        var iceRend = icePatch != null ? icePatch.GetComponent<MeshRenderer>() : null;
        if (iceRend != null) iceSay += ", shader " + iceRend.sharedMaterial.shader.name + ", queue " + iceRend.sharedMaterial.renderQueue + ", surface " + (iceRend.sharedMaterial.HasProperty("_Surface") ? iceRend.sharedMaterial.GetFloat("_Surface").ToString("F0") : "?") + ", colour " + iceRend.sharedMaterial.color + ", enabled " + iceRend.enabled;
        string waterSay = waterPatch == null ? "no water patch" : "water patch verts " + (waterPatch.GetComponent<MeshFilter>() != null && waterPatch.GetComponent<MeshFilter>().sharedMesh != null ? waterPatch.GetComponent<MeshFilter>().sharedMesh.vertexCount.ToString() : "none");
        int frozen = 0, open = 0;
        for (int dx = -12; dx <= 12; dx++) for (int dz = -12; dz <= 12; dz++) { if (WaterSurface.IsFrozen(fx + dx, fz + dz, seed)) frozen++; else if (WaterSurface.IsUnderwater(fx + dx, fz + dz, seed)) open++; }
        string raySay = "ray: nothing";
        if (Physics.Raycast(bank + onto * 6f + Vector3.up * 3f, Vector3.down, out var hit, 20f)) raySay = "ray hit " + hit.collider.name + " at y " + hit.point.y.ToString("F2") + " (level " + WaterSurface.Level.ToString("F2") + ")";
        Mark("ICE CHECK: chunk " + chunkAt + ": " + iceSay + "; " + waterSay + "; tiles within 12: frozen " + frozen + ", open " + open + "; " + raySay);

        cam.transform.position = bank + Vector3.up * 2.4f - onto * 3f;
        cam.transform.LookAt(new Vector3(bank.x, WaterSurface.Level, bank.z) + onto * 12f);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-rr-snow.png"));
        yield return new WaitForSecondsRealtime(0.5f);
        cam.transform.position = bank + onto * 10f + Vector3.up * 28f;
        cam.transform.rotation = Quaternion.LookRotation(Vector3.down, onto);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-rr-above.png"));
        yield return new WaitForSecondsRealtime(0.5f);

        // and out onto the ice
        var inputs = player.GetComponent<StarterAssets.StarterAssetsInputs>();
        var controller = player.GetComponent<StarterAssets.ThirdPersonController>();
        var yawField = typeof(StarterAssets.ThirdPersonController).GetField("_cinemachineTargetYaw", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        if (yawField != null) yawField.SetValue(controller, Quaternion.LookRotation(onto).eulerAngles.y);
        inputs.move = new Vector2(0f, 1f);
        yield return new WaitForSecondsRealtime(4f);
        inputs.move = Vector2.zero;
        yield return new WaitForSecondsRealtime(0.5f);
        int ptx = Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), ptz = Mathf.RoundToInt(player.position.z / WorldGrid.TileSize);
        Mark("ICE: standing on a frozen tile " + WaterSurface.IsFrozen(ptx, ptz, seed) + ", feet " + (player.position.y - WaterSurface.Level).ToString("F2") + " m above the water level, in water " + Swimming.InWater + ", afloat " + Swimming.Afloat);
        cam.transform.position = player.position + player.right * 4f + Vector3.up * 1.6f;
        cam.transform.LookAt(player.position + Vector3.up * 0.6f);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-rr-ice.png"));
        yield return new WaitForSecondsRealtime(0.5f);
        TimeOfDay.Instance.ForceOvercast(-1f);
        Mark("done");
    }
}
