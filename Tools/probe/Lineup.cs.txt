// A probe: every kind in two rows on open ground, photographed once for
// the README.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(6f);

        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }

        Transform player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        string dir = Application.persistentDataPath;
        if (TimeOfDay.Instance != null) TimeOfDay.Instance.SetTime(0.38f);

        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var stateField = typeof(Animal).GetField("state", flags);
        var stateType = typeof(Animal).GetNestedType("State", System.Reflection.BindingFlags.NonPublic);
        var untilField = typeof(Animal).GetField("until", flags);
        var manager = FindFirstObjectByType<Wildlife>();
        foreach (var wild in FindObjectsByType<Animal>(FindObjectsSortMode.None)) Destroy(wild.gameObject);
        manager.enabled = false;

        // open, level ground: the desert, where little grows
        var home = new Vector2Int(Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), Mathf.RoundToInt(player.position.z / WorldGrid.TileSize));
        Vector2Int flat = home; bool found = false;
        for (int r = 2; r < 500 && !found; r += 2)
        for (int dx = -r; dx <= r && !found; dx += 2)
        for (int dz = -r; dz <= r && !found; dz += 2)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
            int cx = home.x + dx, cz = home.y + dz;
            float lo = float.MaxValue, hi = float.MinValue; bool ok = true;
            for (int i = -9; i <= 9 && ok; i++) for (int j = -5; j <= 5 && ok; j++)
            {
                float h = WorldHeight.SurfaceY(cx + i, cz + j, seed); lo = Mathf.Min(lo, h); hi = Mathf.Max(hi, h);
                if (WaterSurface.IsUnderwater(cx + i, cz + j, seed) || Regions.CharacterAtTile(cx + i, cz + j, seed) != Regions.Character.Desert) ok = false;
            }
            if (ok && hi - lo < 0.3f) { flat = new Vector2Int(cx, cz); found = true; }
        }
        Mark("desert lineup ground " + found);
        var origin = new Vector3(flat.x * WorldGrid.TileSize, WorldHeight.SurfaceY(flat.x, flat.y, seed), flat.y * WorldGrid.TileSize);
        body.enabled = false; player.position = origin + new Vector3(0f, 0.3f, -70f); body.enabled = true;
        yield return new WaitForSecondsRealtime(6f);
        HideInterface();
        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;

        var kinds = new[] { FaunaKind.Deer, FaunaKind.Wolf, FaunaKind.Boar, FaunaKind.Goat, FaunaKind.Heron, FaunaKind.Eagle, FaunaKind.Fox, FaunaKind.Hare, FaunaKind.Tortoise,
                            FaunaKind.Rabbit, FaunaKind.Marmot, FaunaKind.Raven, FaunaKind.Owl, FaunaKind.Crab, FaunaKind.Hedgehog, FaunaKind.Frog, FaunaKind.Scorpion, FaunaKind.Bat };
        for (int i = 0; i < kinds.Length; i++)
        {
            int row = i < 9 ? 0 : 1, col = i < 9 ? i : i - 9;
            var at = origin + new Vector3((col - 4f) * 2.6f, 0f, row == 0 ? 3f : -1.5f);
            var go = new GameObject("lineup " + kinds[i]);
            var an = go.AddComponent<Animal>();
            an.Settle(kinds[i], seed, player, at, 1f);
            typeof(Animal).GetField("yaw", flags).SetValue(an, 200f);
            go.transform.rotation = Quaternion.Euler(0f, 200f, 0f);
            stateField.SetValue(an, System.Enum.Parse(stateType, "Stand"));
            untilField.SetValue(an, Time.time + 1000f);
            // the owl on its snag would hide the row: keep it on the ground
            if (kinds[i] == FaunaKind.Owl) { typeof(Animal).GetField("perched", flags).SetValue(an, false); typeof(Animal).GetField("altitude", flags).SetValue(an, 0f); var snag = GameObject.Find("snag"); if (snag != null) Destroy(snag); }
        }
        yield return new WaitForSecondsRealtime(2.5f);
        HideInterface();
        cam.fieldOfView = 46f;
        cam.transform.position = origin + new Vector3(0f, 3.2f, -13.5f);
        cam.transform.rotation = Quaternion.LookRotation((origin + new Vector3(0f, 0.5f, 1f)) - cam.transform.position, Vector3.up);
        yield return new WaitForSecondsRealtime(0.5f);
        ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-lineup.png"));
        yield return new WaitForSecondsRealtime(1f);
        Mark("done");
    }
}
