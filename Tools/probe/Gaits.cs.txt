// A probe: the gaits, measured. A heron walking (steps a second, slip,
// photographed side on); a goat running (how much a hip angle jumps
// between frames, steps a second); an eagle over twenty seconds (how far
// it went, how much it turned, how many wing beats); a heron's fish
// against its beak.
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(6f);

        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }

        Transform player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        string dir = Application.persistentDataPath;
        if (TimeOfDay.Instance != null) TimeOfDay.Instance.SetTime(0.40f);

        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var stateField = typeof(Animal).GetField("state", flags);
        var stateType = typeof(Animal).GetNestedType("State", System.Reflection.BindingFlags.NonPublic);
        var untilField = typeof(Animal).GetField("until", flags);
        var targetField = typeof(Animal).GetField("target", flags);
        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
        var manager = FindFirstObjectByType<Wildlife>();
        foreach (var wild in FindObjectsByType<Animal>(FindObjectsSortMode.None)) Destroy(wild.gameObject);
        manager.enabled = false;

        // open level ground: desert or lowland
        var home = new Vector2Int(Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), Mathf.RoundToInt(player.position.z / WorldGrid.TileSize));
        Vector2Int flat = home; bool found = false;
        for (int r = 2; r < 500 && !found; r += 2)
        for (int dx = -r; dx <= r && !found; dx += 2)
        for (int dz = -r; dz <= r && !found; dz += 2)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
            int cx = home.x + dx, cz = home.y + dz;
            float lo = float.MaxValue, hi = float.MinValue; bool ok = true;
            for (int i = -12; i <= 12 && ok; i++) for (int j = -4; j <= 4 && ok; j++)
            {
                float h = WorldHeight.SurfaceY(cx + i, cz + j, seed); lo = Mathf.Min(lo, h); hi = Mathf.Max(hi, h);
                var k = Regions.CharacterAtTile(cx + i, cz + j, seed);
                if (WaterSurface.IsUnderwater(cx + i, cz + j, seed) || (k != Regions.Character.Desert && k != Regions.Character.Lowland)) ok = false;
            }
            if (ok && hi - lo < 0.01f) { flat = new Vector2Int(cx, cz); found = true; }
        }
        var origin = new Vector3(flat.x * WorldGrid.TileSize, WorldHeight.SurfaceY(flat.x, flat.y, seed), flat.y * WorldGrid.TileSize);
        body.enabled = false; player.position = origin + new Vector3(0f, 0.3f, -75f); body.enabled = true;
        yield return new WaitForSecondsRealtime(5f);
        HideInterface();

        Animal Make(FaunaKind kind, Vector3 at, float yaw)
        {
            var go = new GameObject("probe " + kind);
            var an = go.AddComponent<Animal>();
            an.Settle(kind, seed, player, at, 1f);
            typeof(Animal).GetField("yaw", flags).SetValue(an, yaw);
            go.transform.rotation = Quaternion.Euler(0f, yaw, 0f);
            return an;
        }
        void Send(Animal an, Vector3 to) { stateField.SetValue(an, System.Enum.Parse(stateType, "Wander")); targetField.SetValue(an, to); untilField.SetValue(an, Time.time + 1000f); }
        void Shoot(Animal an, string name, float back, float up)
        {
            var p = an.transform.position;
            cam.transform.position = p + new Vector3(0f, up, -back);
            cam.transform.rotation = Quaternion.LookRotation((p + Vector3.up * up * 0.5f) - cam.transform.position, Vector3.up);
            HideInterface();
            ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-ga-" + name + ".png"));
        }

        // 1. the heron walking
        {
            var h = Make(FaunaKind.Heron, origin + new Vector3(-8f, 0f, 0f), 90f);
            yield return new WaitForSecondsRealtime(0.5f);
            Send(h, origin + new Vector3(30f, 0f, 0f));
            yield return new WaitForSecondsRealtime(2f);
            var was = new bool[4]; int steps = 0; float slip = 0f; var last = new Vector3[4]; var lastRoot = h.transform.position; float travel = 0f;
            float began = Time.realtimeSinceStartup; int shot = 0; float next = Time.realtimeSinceStartup + 0.3f;
            while (Time.realtimeSinceStartup - began < 6f)
            {
                yield return null;
                travel += Vector3.Distance(new Vector3(h.transform.position.x, 0f, h.transform.position.z), new Vector3(lastRoot.x, 0f, lastRoot.z)); lastRoot = h.transform.position;
                for (int i = 2; i < 4; i++)
                {
                    if (h.Planted[i] && !was[i]) steps++;
                    if (h.Planted[i] && was[i]) slip += Vector3.Distance(new Vector3(h.Feet[i].x, 0f, h.Feet[i].z), new Vector3(last[i].x, 0f, last[i].z));
                    was[i] = h.Planted[i]; last[i] = h.Feet[i];
                }
                if (shot < 3 && Time.realtimeSinceStartup > next) { Shoot(h, "heron" + shot, 3.6f, 1.0f); shot++; next += 0.7f; }
            }
            Mark(string.Format("HERON WALK: {0:F2} m in 6 s, {1} footfalls ({2:F2} steps/s), planted feet slid {3:F0}% of travel", travel, steps, steps / 6f, 100f * slip / Mathf.Max(0.01f, travel)));
            Destroy(h.gameObject);
        }

        // 2. the goat running
        {
            var g = Make(FaunaKind.Goat, origin + new Vector3(-12f, 0f, 3f), 90f);
            yield return new WaitForSecondsRealtime(0.5f);
            g.Startle(origin + new Vector3(-40f, 0f, 3f), true);
            yield return new WaitForSecondsRealtime(0.8f);
            var hips = g.transform.Find("frame/foreleg");
            float lastAngle = hips.localEulerAngles.x, worst = 0f, sum = 0f; int frames = 0, steps = 0; var was = new bool[4];
            float lastVel = 0f; int reversals = 0;
            float began = Time.realtimeSinceStartup; int shot = 0; float next = Time.realtimeSinceStartup + 0.2f;
            while (Time.realtimeSinceStartup - began < 3f)
            {
                yield return null;
                float a = hips.localEulerAngles.x; float signed = Mathf.DeltaAngle(lastAngle, a); float d = Mathf.Abs(signed); lastAngle = a;
                worst = Mathf.Max(worst, d); sum += d; frames++;
                if (Mathf.Abs(signed) > 0.05f) { if (Mathf.Sign(signed) != Mathf.Sign(lastVel) && lastVel != 0f) reversals++; lastVel = signed; }
                for (int i = 0; i < 4; i++) { if (g.Planted[i] && !was[i]) steps++; was[i] = g.Planted[i]; }
                if (shot < 2 && Time.realtimeSinceStartup > next) { Shoot(g, "goatrun" + shot, 4.5f, 1.2f); shot++; next += 0.35f; }
            }
            string st = stateField.GetValue(g).ToString();
            Mark(string.Format("GOAT RUN: state {0}, hip swing reversals {1:F1}/s (a clean gallop is about five), worst jump {2:F1} deg/frame over {3} frames; {4:F1} footfalls/s", st, reversals / 3f, worst, frames, steps / 3f));
            Destroy(g.gameObject);
        }

        // 3. the eagle over twenty seconds
        {
            var e = Make(FaunaKind.Eagle, origin, 0f);
            yield return new WaitForSecondsRealtime(8f);
            var wing = e.transform.Find("frame/wing");
            float went = 0f, turned = 0f; var lastP = e.transform.position; float lastYaw = e.transform.eulerAngles.y;
            float lastWing = (wing.localRotation * Vector3.back).y, lastVel = 0f; int beats = 0; float lo = 999f, hi = -999f;
            float began = Time.realtimeSinceStartup; bool shotIt = false;
            while (Time.realtimeSinceStartup - began < 20f)
            {
                yield return null;
                went += Vector3.Distance(e.transform.position, lastP); lastP = e.transform.position;
                turned += Mathf.Abs(Mathf.DeltaAngle(e.transform.eulerAngles.y, lastYaw)); lastYaw = e.transform.eulerAngles.y;
                float w = (wing.localRotation * Vector3.back).y; float vel = w - lastWing; lastWing = w;
                if (Mathf.Abs(vel) > 0.0015f) { if (Mathf.Sign(vel) != Mathf.Sign(lastVel) && lastVel != 0f) beats++; lastVel = vel; }
                float alt = (float)typeof(Animal).GetField("altitude", flags).GetValue(e); lo = Mathf.Min(lo, alt); hi = Mathf.Max(hi, alt);
                if (!shotIt && Time.realtimeSinceStartup - began > 10f)
                {
                    var p = e.transform.position + Vector3.up * alt;
                    cam.transform.position = p - e.transform.forward * 6f + Vector3.up * 3.5f + e.transform.right * 2f;
                    cam.transform.rotation = Quaternion.LookRotation((p + e.transform.forward * 2f - Vector3.up * 1.5f) - cam.transform.position, Vector3.up);
                    HideInterface(); ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-ga-eagle.png")); shotIt = true;
                }
            }
            Mark(string.Format("EAGLE: went {0:F0} m in 20 s, turned {1:F0} deg in all, altitude {2:F1}..{3:F1}, wing reversals {4} (a beat is two)", went, turned, lo, hi, beats));
            Destroy(e.gameObject);
        }

        // 4. the fish against the beak
        {
            var h = Make(FaunaKind.Heron, origin + new Vector3(4f, 0f, 0f), 200f);
            stateField.SetValue(h, System.Enum.Parse(stateType, "Graze")); untilField.SetValue(h, Time.time + 100f);
            yield return new WaitForSecondsRealtime(1f);
            var head = h.transform.Find("frame/head");
            var fish = AnimalBuilder.FishInBeak(head);
            fish.localPosition = new Vector3(0f, 0.455f * 1.05f, 0.26f * 1.05f); fish.localRotation = Quaternion.Euler(0f, 90f, 12f);
            typeof(Animal).GetField("caught", flags).SetValue(h, fish);
            typeof(Animal).GetField("caughtUntil", flags).SetValue(h, Time.time + 8f);
            yield return new WaitForSecondsRealtime(1.2f);
            var beakMid = head.TransformPoint(new Vector3(0f, 0.455f * 1.05f, 0.26f * 1.05f));
            var bounds = new Bounds(fish.position, Vector3.zero); bool any = false;
            foreach (var r in fish.GetComponentsInChildren<Renderer>()) { if (!any) { bounds = r.bounds; any = true; } else bounds.Encapsulate(r.bounds); }
            Mark(string.Format("FISH: fish centre is {0:F3} m from the beak's middle ({1:F3} m above it)", Vector3.Distance(bounds.center, beakMid), bounds.center.y - beakMid.y));
            var p = head.TransformPoint(new Vector3(0f, 0.45f * 1.05f, 0.2f * 1.05f));
            cam.transform.position = p + new Vector3(1.4f, 0.4f, -1.1f);
            cam.transform.rotation = Quaternion.LookRotation(p - cam.transform.position, Vector3.up);
            HideInterface(); yield return new WaitForSecondsRealtime(0.2f);
            ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-ga-fish.png"));
            yield return new WaitForSecondsRealtime(0.8f);
            Destroy(h.gameObject);
        }

        Mark("done");
    }
}
