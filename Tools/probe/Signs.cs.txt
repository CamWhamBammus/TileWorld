// A probe: what the animals leave. A deer walked across snow and the
// prints counted and photographed; a boar left rooting for a while; a
// heron sent up off the ground; a deer walked over grass, which should
// leave nothing.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(6f);

        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }

        Transform player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        string dir = Application.persistentDataPath;
        if (TimeOfDay.Instance != null) TimeOfDay.Instance.SetTime(0.40f);

        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var stateField = typeof(Animal).GetField("state", flags);
        var stateType = typeof(Animal).GetNestedType("State", System.Reflection.BindingFlags.NonPublic);
        var untilField = typeof(Animal).GetField("until", flags);
        var targetField = typeof(Animal).GetField("target", flags);
        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;

        var manager = FindFirstObjectByType<Wildlife>();
        foreach (var wild in FindObjectsByType<Animal>(FindObjectsSortMode.None)) Destroy(wild.gameObject);
        manager.enabled = false;

        int Marks()
        {
            var tracks = FindFirstObjectByType<Tracks>();
            if (tracks == null) return -1;
            var list = (System.Collections.IList)typeof(Tracks).GetField("marks", flags).GetValue(tracks);
            return list.Count;
        }

        Vector2Int Level(System.Func<int, int, bool> suits, int across, int deep)
        {
            var home = new Vector2Int(Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), Mathf.RoundToInt(player.position.z / WorldGrid.TileSize));
            for (int r = 2; r < 400; r += 2)
            for (int dx = -r; dx <= r; dx += 2)
            for (int dz = -r; dz <= r; dz += 2)
            {
                if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
                int cx = home.x + dx, cz = home.y + dz;
                float lo = float.MaxValue, hi = float.MinValue; bool ok = true;
                for (int i = -across; i <= across && ok; i++) for (int j = -deep; j <= deep && ok; j++)
                {
                    float h = WorldHeight.SurfaceY(cx + i, cz + j, seed); lo = Mathf.Min(lo, h); hi = Mathf.Max(hi, h);
                    if (WaterSurface.IsUnderwater(cx + i, cz + j, seed) || !suits(cx + i, cz + j)) ok = false;
                }
                if (ok && hi - lo < 0.3f) return new Vector2Int(cx, cz);
            }
            return home;
        }

        IEnumerator Walk(FaunaKind kind, Vector2Int at, string name, float seconds)
        {
            var origin = new Vector3(at.x * WorldGrid.TileSize, WorldHeight.SurfaceY(at.x, at.y, seed), at.y * WorldGrid.TileSize);
            body.enabled = false; player.position = origin + new Vector3(0f, 0.3f, -70f); body.enabled = true;
            yield return new WaitForSecondsRealtime(5f);
            HideInterface();
            int before = Marks();
            var go = new GameObject("probe " + kind);
            var an = go.AddComponent<Animal>();
            an.Settle(kind, seed, player, origin + new Vector3(-6f, 0f, 0f));
            typeof(Animal).GetField("yaw", flags).SetValue(an, 90f);
            go.transform.rotation = Quaternion.Euler(0f, 90f, 0f);
            yield return new WaitForSecondsRealtime(0.5f);
            stateField.SetValue(an, System.Enum.Parse(stateType, "Wander"));
            targetField.SetValue(an, origin + new Vector3(30f, 0f, 0f));
            untilField.SetValue(an, Time.time + 1000f);
            yield return new WaitForSecondsRealtime(seconds);
            int after = Marks();
            Mark(name + ": marks before " + before + " after " + after + " (" + (after - before) + " new) over " + Vector3.Distance(an.transform.position, origin + new Vector3(-6f, 0f, 0f)).ToString("F1") + " m");
            var p = an.transform.position;
            cam.transform.position = p + new Vector3(-5f, 2.4f, -3.5f);
            cam.transform.rotation = Quaternion.LookRotation((p + new Vector3(-2f, 0f, 0f)) - cam.transform.position, Vector3.up);
            HideInterface();
            yield return new WaitForSecondsRealtime(0.3f);
            ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-si-" + name + ".png"));
            yield return new WaitForSecondsRealtime(0.8f);
            Destroy(go);
        }

        // 1. snow
        var snow = Level((x, z) => SnowCover.IsSnowy(x, z, seed), 10, 3);
        Mark("snow ground at " + snow);
        yield return Walk(FaunaKind.Deer, snow, "snow", 5f);

        // 2. grass: nothing
        var grass = Level((x, z) => !SnowCover.IsSnowy(x, z, seed) && Regions.CharacterAtTile(x, z, seed) == Regions.Character.Forest, 10, 3);
        yield return Walk(FaunaKind.Deer, grass, "grass", 4f);

        // 3. a boar rooting
        {
            var origin = new Vector3(grass.x * WorldGrid.TileSize, WorldHeight.SurfaceY(grass.x, grass.y, seed), grass.y * WorldGrid.TileSize);
            int before = Marks();
            var go = new GameObject("probe boar");
            var an = go.AddComponent<Animal>();
            an.Settle(FaunaKind.Boar, seed, player, origin);
            stateField.SetValue(an, System.Enum.Parse(stateType, "Graze"));
            untilField.SetValue(an, Time.time + 1000f);
            yield return new WaitForSecondsRealtime(12f);
            Mark("rooting: " + (Marks() - before) + " patches in 12 s");
            var p = an.transform.position;
            cam.transform.position = p + new Vector3(2.5f, 1.8f, -3.5f);
            cam.transform.rotation = Quaternion.LookRotation(p - cam.transform.position, Vector3.up);
            HideInterface();
            yield return new WaitForSecondsRealtime(0.3f);
            ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-si-rooting.png"));
            yield return new WaitForSecondsRealtime(0.8f);
            Destroy(go);
        }

        // 4. a heron going up
        {
            var origin = new Vector3(grass.x * WorldGrid.TileSize, WorldHeight.SurfaceY(grass.x, grass.y, seed), grass.y * WorldGrid.TileSize) + new Vector3(4f, 0f, 4f);
            int before = Marks();
            var go = new GameObject("probe heron");
            var an = go.AddComponent<Animal>();
            an.Settle(FaunaKind.Heron, seed, player, origin);
            yield return new WaitForSecondsRealtime(0.8f);
            an.Startle(origin + new Vector3(0f, 0f, -10f), true);
            yield return new WaitForSecondsRealtime(1.5f);
            Mark("feather: " + (Marks() - before) + " left where the heron stood");
            cam.transform.position = origin + new Vector3(1.2f, 1.2f, -2.2f);
            cam.transform.rotation = Quaternion.LookRotation(origin - cam.transform.position, Vector3.up);
            HideInterface();
            yield return new WaitForSecondsRealtime(0.3f);
            ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-si-feather.png"));
            yield return new WaitForSecondsRealtime(0.8f);
            Destroy(go);
        }

        Mark("done");
    }
}
