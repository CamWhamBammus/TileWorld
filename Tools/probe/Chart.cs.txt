// A probe for the chart: surveys a wide square about the player so there is something to draw,
// opens the map and photographs it. The point is whether the countries can be told apart on it.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        var save = WorldLibrary.Create("Probe chart", 5, true, true, 0.42f, 20f, true, true);
        WorldLibrary.Enter(save);
        yield return new WaitForSecondsRealtime(9f);
        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var arrival = FindFirstObjectByType<Arrival>();
        if (arrival != null) typeof(Arrival).GetMethod("Close", flags).Invoke(arrival, null);
        TimeOfDay.Instance.SetTime(0.42f); TimeOfDay.Instance.Paused = true;
        yield return new WaitForSecondsRealtime(4f);

        var world = FindFirstObjectByType<ChunkManager>();
        var centre = WorldGrid.WorldToChunk(world.PlayerTransform.position);

        // survey a wide square, which is what a chart needs before it is worth looking at
        int drawn = 0;
        for (int dx = -30; dx <= 30; dx++)
        for (int dz = -30; dz <= 30; dz++)
            if (ExplorationLog.Survey(new Vector2Int(centre.x + dx, centre.y + dz))) drawn++;
        Mark("CHART surveyed " + drawn + " chunks about " + centre);

        // what countries are in reach, so the picture can be read against a list
        var seen = new System.Collections.Generic.SortedDictionary<string, int>();
        for (int dx = -30; dx <= 30; dx += 2)
        for (int dz = -30; dz <= 30; dz += 2)
        {
            string who = Regions.CharacterAt(new Vector2Int(centre.x + dx, centre.y + dz), world.WorldSeed).ToString();
            seen[who] = (seen.TryGetValue(who, out int n) ? n : 0) + 1;
        }
        var said = new System.Text.StringBuilder();
        foreach (var pair in seen) said.Append(" " + pair.Key + ":" + pair.Value);
        Mark("CHART countries in the square:" + said);

        yield return new WaitForSecondsRealtime(1f);
        // The map opens off a key in its own Update, so it is opened here the way that does:
        // set the flag, show the panel, mark it dirty, tell the screens.
        var map = FindFirstObjectByType<WorldMap>();
        var panel = (GameObject)typeof(WorldMap).GetField("panel", flags).GetValue(map);
        typeof(WorldMap).GetField("open", flags).SetValue(map, true);
        typeof(WorldMap).GetField("dirty", flags).SetValue(map, true);
        panel.SetActive(true);
        ScreenState.Open(ScreenState.Screen.Map);
        Mark("CHART map panel up: " + panel.activeInHierarchy);
        yield return new WaitForSecondsRealtime(4f);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-ch-map.png"));
        yield return new WaitForSecondsRealtime(1.5f);
        Mark("done");
        WorldLibrary.Delete(save.id);
    }
}
