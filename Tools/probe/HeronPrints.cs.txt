// A probe: a heron walking, side on, four frames; a heron in the shallows;
// a deer on grass; prints on sand from low down.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(6f);
        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }
        Transform player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        string dir = Application.persistentDataPath;
        if (TimeOfDay.Instance != null) TimeOfDay.Instance.SetTime(0.40f);
        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var stateField = typeof(Animal).GetField("state", flags);
        var stateType = typeof(Animal).GetNestedType("State", System.Reflection.BindingFlags.NonPublic);
        var untilField = typeof(Animal).GetField("until", flags);
        var targetField = typeof(Animal).GetField("target", flags);
        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
        FindFirstObjectByType<Wildlife>().enabled = false;
        foreach (var wild in FindObjectsByType<Animal>(FindObjectsSortMode.None)) Destroy(wild.gameObject);

        // how many tall things the undergrowth has standing within reach of a point
        int TallNear(Vector3 at, float within)
        {
            int count = 0;
            try
            {
                var under = FindFirstObjectByType<Undergrowth>();
                var every = (Flora.Sprout[])typeof(Undergrowth).GetField("every", flags).GetValue(under);
                var gathered = (Matrix4x4[][])typeof(Undergrowth).GetField("gathered", flags).GetValue(under);
                for (int i = 0; every != null && gathered != null && i < every.Length && i < gathered.Length; i++)
                {
                    if (gathered[i] == null) continue;
                    foreach (var m in gathered[i])
                    {
                        if (m.lossyScale.y * every[i].Size < 1.4f) continue;
                        var pos = m.GetPosition(); pos.y = 0f; var flat = at; flat.y = 0f;
                        if (Vector3.Distance(pos, flat) < within) count++;
                    }
                }
            }
            catch { }
            return count;
        }

        Vector2Int Level(System.Func<int, int, bool> suits, int across, int deep)
        {
            var home = new Vector2Int(Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), Mathf.RoundToInt(player.position.z / WorldGrid.TileSize));
            for (int r = 2; r < 56; r += 2)
            for (int dx = -r; dx <= r; dx += 2)
            for (int dz = -r; dz <= r; dz += 2)
            {
                if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
                int cx = home.x + dx, cz = home.y + dz;
                float lo = float.MaxValue, hi = float.MinValue; bool ok = true;
                for (int i = -across; i <= across && ok; i++) for (int j = -deep; j <= deep && ok; j++)
                {
                    float h = WorldHeight.SurfaceY(cx + i, cz + j, seed); lo = Mathf.Min(lo, h); hi = Mathf.Max(hi, h);
                    if (!suits(cx + i, cz + j)) ok = false;
                }
                if (ok && hi - lo < 0.26f && TallNear(new Vector3(cx * 2f, 0f, cz * 2f), 9f) == 0) return new Vector2Int(cx, cz);
            }
            Mark("SITE: nothing suitable within range; falling back to home, which may be wrong");
            return home;
        }

        IEnumerator Frames(FaunaKind kind, Vector2Int at, string name, bool walk, float camUp, float camBack)
        {
            var origin = new Vector3(at.x * WorldGrid.TileSize, WorldHeight.SurfaceY(at.x, at.y, seed), at.y * WorldGrid.TileSize);
            body.enabled = false; player.position = origin + new Vector3(0f, 0.3f, -70f); body.enabled = true;
            yield return new WaitForSecondsRealtime(5f);
            HideInterface();
            var go = new GameObject("probe " + kind);
            var an = go.AddComponent<Animal>();
            an.Settle(kind, seed, player, origin + new Vector3(-4f, 0f, 0f), 1f);
            typeof(Animal).GetField("yaw", flags).SetValue(an, 90f);
            go.transform.rotation = Quaternion.Euler(0f, 90f, 0f);
            yield return new WaitForSecondsRealtime(0.8f);
            if (walk)
            {
                stateField.SetValue(an, System.Enum.Parse(stateType, kind == FaunaKind.Goat ? "Flee" : "Wander"));
                targetField.SetValue(an, origin + new Vector3(30f, 0f, 0f));
            }
            else { stateField.SetValue(an, System.Enum.Parse(stateType, "Stand")); }
            untilField.SetValue(an, Time.time + 1000f);
            yield return new WaitForSecondsRealtime(1.2f);
            // the camera on whichever side is dry ground, not over a lake
            var probeAt = an.transform.position + new Vector3(0f, 0f, -camBack);
            float side = WaterSurface.IsUnderwater(Mathf.RoundToInt(probeAt.x / 2f), Mathf.RoundToInt(probeAt.z / 2f), seed) ? 1f : -1f;
            for (int f = 0; f < 6; f++)
            {
                var p = an.transform.position;
                cam.transform.position = p + new Vector3(0f, camUp, side * camBack);
                cam.transform.rotation = Quaternion.LookRotation((p + Vector3.up * camUp * 0.5f) - cam.transform.position, Vector3.up);
                HideInterface();
                yield return null;
                ScreenCapture.CaptureScreenshot(Path.Combine(dir, "probe-hp-" + name + f + ".png"));
                yield return new WaitForSecondsRealtime(walk ? (kind == FaunaKind.Goat ? 0.12f : 0.3f) : 0.1f);
                if (!walk) break;
            }
            string feet = ""; for (int i = 0; i < 4; i++) feet += (an.Feet[i].y - origin.y).ToString("F2") + " ";
            Mark(name + ": feet above walking surface " + feet + "| root " + (an.transform.position.y - origin.y).ToString("F2"));
            Destroy(go);
        }

        // first, somewhere open by the map alone: a lowland chunk, dry and flat
        // at its middle; go there and let it draw, so trees can be counted
        {
            var homeChunk = WorldGrid.WorldToChunk(player.position);
            bool got = false;
            for (int r = 1; r < 60 && !got; r++)
            for (int dx = -r; dx <= r && !got; dx++)
            for (int dz = -r; dz <= r && !got; dz++)
            {
                if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
                var chunk = new Vector2Int(homeChunk.x + dx, homeChunk.y + dz);
                if (Regions.CharacterAt(chunk, seed) != Regions.Character.Lowland) continue;
                int cx = chunk.x * WorldGrid.TilesPerChunk + 7, cz = chunk.y * WorldGrid.TilesPerChunk + 7;
                if (WaterSurface.IsUnderwater(cx, cz, seed) || SnowCover.IsSnowy(cx, cz, seed)) continue;
                var c = WorldGrid.ChunkCenter(chunk);
                body.enabled = false; player.position = new Vector3(c.x, WorldHeight.SurfaceY(cx, cz, seed) + 1f, c.z); body.enabled = true;
                got = true;
            }
            Mark("lowland chunk found " + got);
            yield return new WaitForSecondsRealtime(8f);
        }

        var grass = Level((x, z) => !WaterSurface.IsUnderwater(x, z, seed) && !SnowCover.IsSnowy(x, z, seed)
            && (Regions.CharacterAtTile(x, z, seed) == Regions.Character.Forest || Regions.CharacterAtTile(x, z, seed) == Regions.Character.Lowland || Regions.CharacterAtTile(x, z, seed) == Regions.Character.Hills), 5, 2);
        Mark(string.Format("SITE {0}: surface {1:F2}, water level {2:F2}, underwater {3}, snowy {4}, region {5}, character of chunk {6}, footing {7:F2}",
            grass, WorldHeight.SurfaceY(grass.x, grass.y, seed), WaterSurface.Level, WaterSurface.IsUnderwater(grass.x, grass.y, seed),
            SnowCover.IsSnowy(grass.x, grass.y, seed), Regions.CharacterAtTile(grass.x, grass.y, seed),
            Regions.CharacterAt(WorldGrid.WorldToChunk(new Vector3(grass.x * 2f, 0f, grass.y * 2f)), seed), Animal.FootingAt(grass.x, grass.y, seed)));
        yield return Frames(FaunaKind.Heron, grass, "heronwalk", true, 1.0f, 4.2f);
        yield return Frames(FaunaKind.Deer, grass, "deerstand", false, 0.9f, 4.5f);
        yield return Frames(FaunaKind.Goat, grass, "goatrun", true, 0.9f, 5.5f);
        Mark("done");
    }
}
