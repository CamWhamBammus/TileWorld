// A probe: how long a save takes, and how long a frame with one in it is.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private float worst; private bool watching;
    private void Update() { if (watching && Time.unscaledDeltaTime * 1000f > worst) worst = Time.unscaledDeltaTime * 1000f; }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        if (TitleMenu.IsUp)
        {
            var worlds = WorldLibrary.All();
            WorldLibrary.Enter(worlds.Count > 0 ? worlds[0] : WorldLibrary.Create("SaveTime", 7777));
            yield return new WaitForSecondsRealtime(6f);
        }
        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }
        yield return new WaitForSecondsRealtime(8f);
        var saver = FindFirstObjectByType<SaveCoordinator>();
        if (saver == null) { Mark("no SaveCoordinator"); Mark("done"); yield break; }

        for (int i = 0; i < 3; i++)
        {
            var sw = System.Diagnostics.Stopwatch.StartNew();
            saver.SaveNow();
            sw.Stop();
            Mark("SAVE " + i + ": " + sw.Elapsed.TotalMilliseconds.ToString("F1") + " ms");
            yield return new WaitForSecondsRealtime(1f);
        }

        // and the worst frame over a long stand, with the time it came
        watching = true; worst = 0f;
        float t0 = Time.realtimeSinceStartup; float lastWorst = 0f;
        while (Time.realtimeSinceStartup - t0 < 75f)
        {
            if (worst > lastWorst + 20f) { Mark("WORST so far " + worst.ToString("F0") + " ms at " + (Time.realtimeSinceStartup - t0).ToString("F0") + " s"); lastWorst = worst; }
            yield return null;
        }
        Mark("worst frame in 75 s standing: " + worst.ToString("F0") + " ms");
        Mark("done");
    }
}
