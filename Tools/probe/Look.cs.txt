// A probe for the picture: the same shore at noon with and without the
// grade, at dusk, at night and in the rain, the water close to with the
// surveyor wading in it, and a frozen lake in the snow. Says whether the
// camera is post-processing, what the grade is doing, which shader the
// water has, and whether any shot has the magenta of a shader that failed.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";
    private Camera cam; private Transform player;

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        Mark("boot");
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static void HideInterface()
    {
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
            if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
    }

    // the main camera itself is moved, so the grade and the post-processing are in the shot
    private IEnumerator Shot(string name, Vector3 at, Vector3 lookAt)
    {
        cam.transform.position = at;
        cam.transform.LookAt(lookAt);
        yield return null;
        cam.transform.position = at;
        cam.transform.LookAt(lookAt);
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-lk-" + name + ".png"));
        var g = Grading.Instance;
        Mark(name + ": post-processing " + (g != null && g.PostProcessing) + (Grading.Enabled ? ", grade on" : ", grade OFF") + (g != null ? ", exposure " + g.Exposure.ToString("F2") + ", saturation " + g.Saturation.ToString("F0") + ", temperature " + g.Temperature.ToString("F0") : "")
             + ", overcast " + TimeOfDay.Instance.Overcast.ToString("F2") + ", time " + TimeOfDay.Instance.Normalized.ToString("F2"));
        yield return new WaitForSecondsRealtime(0.4f);
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        if (TitleMenu.IsUp)
        {
            var worlds = WorldLibrary.All();
            WorldLibrary.Enter(worlds.Count > 0 ? worlds[0] : WorldLibrary.Create("Look", 7777));
            yield return new WaitForSecondsRealtime(6f);
        }
        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }
        player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        yield return new WaitForSecondsRealtime(4f);
        HideInterface();

        // the water's shader
        var waterGo = GameObject.Find("Water (runtime)");
        string shaderName = "no water patches";
        if (waterGo != null) foreach (Transform t in waterGo.transform) { var r = t.GetComponent<MeshRenderer>(); if (r != null) { shaderName = r.sharedMaterial.shader.name + " (supported " + r.sharedMaterial.shader.isSupported + ")"; break; } }
        Mark("WATER SHADER: " + shaderName);

        // a shore with open water, the same search as the splash probe
        int px = Mathf.RoundToInt(player.position.x / WorldGrid.TileSize), pz = Mathf.RoundToInt(player.position.z / WorldGrid.TileSize);
        int bestX = 0, bestZ = 0, bestDx = 0, bestDz = 0; bool found = false;
        int[] dirs = { 1, 0, -1, 0, 0, 1, 0, -1 };
        for (int r = 4; r < 110 && !found; r += 2)
        for (int dx = -r; dx <= r && !found; dx += 2)
        for (int dz = -r; dz <= r && !found; dz += 2)
        {
            if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
            int tx = px + dx, tz = pz + dz;
            if (WaterSurface.IsUnderwater(tx, tz, seed)) continue;
            for (int d = 0; d < 4 && !found; d++)
            {
                bool deep = false; int wetCount = 0;
                for (int k = 1; k <= 14; k++)
                {
                    int sx = tx + dirs[d * 2] * k, sz = tz + dirs[d * 2 + 1] * k;
                    if (!WaterSurface.IsOpenWater(sx, sz, seed)) { if (k > 2) break; else continue; }
                    wetCount++;
                    if (WaterSurface.Level - WorldHeight.SurfaceY(sx, sz, seed) > 1.6f) { deep = true; break; }
                }
                if (deep && wetCount >= 3) { bestX = tx; bestZ = tz; bestDx = dirs[d * 2]; bestDz = dirs[d * 2 + 1]; found = true; }
            }
        }
        if (!found)
        {
            // any open shore then, deep or not
            for (int r = 4; r < 140 && !found; r += 2)
            for (int dx = -r; dx <= r && !found; dx += 2)
            for (int dz = -r; dz <= r && !found; dz += 2)
            {
                if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
                int tx = px + dx, tz = pz + dz;
                if (WaterSurface.IsUnderwater(tx, tz, seed)) continue;
                for (int d = 0; d < 4 && !found; d++)
                {
                    int wet = 0;
                    for (int k = 2; k <= 8; k++) if (WaterSurface.IsOpenWater(tx + dirs[d * 2] * k, tz + dirs[d * 2 + 1] * k, seed)) wet++;
                    if (wet >= 6) { bestX = tx; bestZ = tz; bestDx = dirs[d * 2]; bestDz = dirs[d * 2 + 1]; found = true; }
                }
            }
        }
        Mark("START at tile " + px + "," + pz + " (" + player.position.ToString("F0") + "), shore found " + found + (found ? " at " + bestX + "," + bestZ : ""));
        if (!found) { Mark("no shore"); Mark("done"); yield break; }
        Vector3 shore = new Vector3(bestX * WorldGrid.TileSize, WorldHeight.SurfaceY(bestX, bestZ, seed) + 1f, bestZ * WorldGrid.TileSize);
        Vector3 into = new Vector3(bestDx, 0f, bestDz);

        // the player stands in the shallows, so the foam has something to break on
        Vector3 wade = shore + into * 5f; wade.y = WaterSurface.Level + 0.8f;
        body.enabled = false; player.position = wade; player.rotation = Quaternion.LookRotation(into); body.enabled = true;
        yield return new WaitForSecondsRealtime(3f);

        cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name.Contains("Cinemachine") || beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
        TimeOfDay.Instance.Paused = true;

        Vector3 wide = shore + Vector3.up * 3.2f - into * 6f; Vector3 wideAt = new Vector3(shore.x, WaterSurface.Level, shore.z) + into * 14f;
        Vector3 close = shore + into * 1.5f + Vector3.up * 1.6f + Quaternion.Euler(0f, 70f, 0f) * into * 2.5f; Vector3 closeAt = wade + Vector3.up * 0.3f;

        TimeOfDay.Instance.ForceOvercast(0f);
        TimeOfDay.Instance.SetTime(0.5f); yield return new WaitForSecondsRealtime(1f);
        yield return Shot("noon", wide, wideAt);
        Grading.Enabled = false; yield return new WaitForSecondsRealtime(0.5f);
        yield return Shot("noon-ungraded", wide, wideAt);
        Grading.Enabled = true; yield return new WaitForSecondsRealtime(0.5f);
        yield return Shot("water-close", close, closeAt);
        TimeOfDay.Instance.SetTime(0.74f); yield return new WaitForSecondsRealtime(1f);
        yield return Shot("dusk", wide, wideAt);
        TimeOfDay.Instance.SetTime(0.95f); yield return new WaitForSecondsRealtime(1f);
        yield return Shot("night", wide, wideAt);
        TimeOfDay.Instance.SetTime(0.5f); TimeOfDay.Instance.ForceOvercast(1f); yield return new WaitForSecondsRealtime(3f);
        yield return Shot("rain", wide, wideAt);
        TimeOfDay.Instance.ForceOvercast(0f);

        // the snow country, by the dev tools
        var tools = FindFirstObjectByType<DevTools>();
        var goFrozen = typeof(DevTools).GetMethod("GoToFrozen", System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance);
        if (tools != null && goFrozen != null)
        {
            goFrozen.Invoke(tools, null);
            yield return new WaitForSecondsRealtime(6f);
            Vector3 bank = player.position;
            yield return Shot("snow", bank + Vector3.up * 3f - player.forward * 4f, bank + player.forward * 12f);
        }

        TimeOfDay.Instance.ForceOvercast(-1f);
        TimeOfDay.Instance.Paused = false;
        Mark("done");
    }
}
