// A probe: the title with the country live behind it. Reads the curtain, the sound and the
// falling name early and late, times the first wave, checks what runs under the title, looks
// at it at all four hours, works the keys, photographs it, enters a world and comes back.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private IEnumerator Shot(string name)
    {
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-tl-" + name + ".png"));
        yield return new WaitForSecondsRealtime(0.3f);
    }

    private string State()
    {
        var world = FindFirstObjectByType<ChunkManager>();
        var title = FindFirstObjectByType<TitleMenu>();
        var cam = Camera.main;
        return "title " + TitleMenu.IsUp + " | menu " + (title != null) + " | logo " + (title != null && title.Logo != null && title.Logo.Ready)
             + " | world " + (world != null ? world.WorldSeed + " radius " + world.ViewRadius : "none")
             + " | player active " + (world != null && world.PlayerTransform != null && world.PlayerTransform.gameObject.activeSelf)
             + " | clouds " + (FindFirstObjectByType<Clouds>() != null) + " | grading " + (Grading.Instance != null && Grading.Instance.PostProcessing)
             + " | surf " + (FindFirstObjectByType<SurfSound>() != null) + " | compass " + (FindFirstObjectByType<CompassBar>() != null)
             + " | surveyor " + (FindFirstObjectByType<Surveyor>() != null) + " | notices " + (FindFirstObjectByType<Notices>() != null)
             + " | cursor " + Cursor.lockState + "/" + Cursor.visible
             + " | cam " + (cam != null ? cam.transform.position.ToString("F1") + " far " + cam.farClipPlane : "-")
             + " | hour " + (TimeOfDay.Instance != null ? TimeOfDay.Instance.Normalized.ToString("F2") + (TimeOfDay.Instance.Paused ? " held" : " turning") : "-")
             + " | scene " + UnityEngine.SceneManagement.SceneManager.GetActiveScene().name;
    }

    private string Early(TitleMenu title)
    {
        return "curtain " + (title != null ? title.CurtainAlpha.ToString("F2") : "-") + " | volume " + AudioListener.volume.ToString("F2") + " of " + Settings.Volume.ToString("F2")
             + " | name landed " + (title != null && title.Logo != null ? title.Logo.Landed.ToString("F2") : "-");
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(0.5f);
        var title = FindFirstObjectByType<TitleMenu>();
        Mark("EARLY 0.5 s: " + Early(title));
        yield return new WaitForSecondsRealtime(1.2f);
        Mark("EARLY 1.7 s: " + Early(title));

        // the first wave over the strand tile the title looks from
        TitleMenu.Viewpoint(TitleMenu.Seed, out _, out _, out _, out var strand, out _);
        Vector3 strandAt = new Vector3(strand.x * WorldGrid.TileSize, WaterSurface.Level, strand.y * WorldGrid.TileSize);
        float firstWave = -1f; float t0 = Time.realtimeSinceStartup;
        for (float t = 0f; t < 14f; t += 0.25f)
        {
            if (Surf.Covered(strandAt, TitleMenu.Seed)) { firstWave = Time.realtimeSinceStartup - t0 + 1.7f; break; }
            yield return new WaitForSecondsRealtime(0.25f);
        }
        Mark("WAVE first over the strand at " + (firstWave < 0f ? "never" : firstWave.ToString("F1") + " s after the start"));
        yield return new WaitForSecondsRealtime(Mathf.Max(0f, 16f - (Time.realtimeSinceStartup - t0) - 1.7f));

        title = FindFirstObjectByType<TitleMenu>();
        title.Look("afternoon");
        yield return new WaitForSecondsRealtime(1f);
        Mark("TITLE: " + State());
        Mark("LATE: " + Early(title) + " | when " + title.When + " | pad voices " + (FindFirstObjectByType<TitlePad>() != null ? FindFirstObjectByType<TitlePad>().Sounding : -1)
             + " | version " + (GameObject.Find("Version") != null) + " | camera off rest " + Quaternion.Angle(Camera.main.transform.rotation, title.BaseRotation).ToString("F2") + " deg");
        var near = new System.Text.StringBuilder("LIFE within 30 m:");
        foreach (var a in Wildlife.Near(Camera.main.transform.position, 30f)) near.Append(" " + a.Kind + "@" + Vector3.Distance(a.transform.position, Camera.main.transform.position).ToString("F0") + "m");
        Mark(near.ToString());

        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var play = (TMPro.TMP_Text)typeof(TitleMenu).GetField("playLabel", flags).GetValue(title);
        string before = play.text;
        typeof(TitleMenu).GetMethod("Choose", flags).Invoke(title, new object[] { 1 });
        Mark("KEYS: play said '" + before + "', after down '" + play.text + "'");
        typeof(TitleMenu).GetMethod("Choose", flags).Invoke(title, new object[] { -1 });

        // the ten more
        var surf = FindFirstObjectByType<SurfSound>();
        var cardGroup = (CanvasGroup)typeof(TitleMenu).GetField("cardGroup", flags).GetValue(title);
        float overcastThen = TimeOfDay.Instance.Overcast;
        Mark("MORE: surf loop at " + surf.ClipTime.ToString("F1") + " s, wanted " + surf.WantedTime.ToString("F1") + " | gulls so far " + BirdSong.Gulls + " | boot prints " + Tracks.Boots + " | card alpha " + cardGroup.alpha.ToString("F2") + " | menu sounds " + MenuSounds.Played);
        MenuSounds.Tick(); MenuSounds.Tock();
        var forget = (TMPro.TMP_Text)typeof(TitleMenu).GetField("forgetLabel", flags).GetValue(title);
        typeof(TitleMenu).GetMethod("Forget", flags).Invoke(title, null);
        string asks = forget.text;
        typeof(TitleMenu).GetField("pendingForget", flags).SetValue(title, null);
        typeof(TitleMenu).GetMethod("Refresh", flags).Invoke(title, null);
        typeof(TitleMenu).GetMethod("Escape", flags).Invoke(title, null);
        var escapeHint = (TMPro.TMP_Text)typeof(TitleMenu).GetField("escapeHint", flags).GetValue(title);
        Mark("MORE: delete asks '" + asks + "' | escape hint showing " + escapeHint.gameObject.activeSelf + " | menu sounds now " + MenuSounds.Played);
        yield return new WaitForSecondsRealtime(3f);

        // a long list: seven more worlds for a moment, the choice walked to the bottom
        var extras = new System.Collections.Generic.List<WorldSave>();
        for (int i = 0; i < 7; i++) extras.Add(WorldLibrary.Create("Probe list " + i, 100 + i));
        typeof(TitleMenu).GetMethod("Refresh", flags).Invoke(title, null);
        int offsetBefore = title.Offset;
        for (int i = 0; i < 9; i++) typeof(TitleMenu).GetMethod("Choose", flags).Invoke(title, new object[] { 1 });
        bool above = GameObject.Find("Above") != null;
        Mark("LIST: " + WorldLibrary.All().Count + " worlds, offset " + offsetBefore + " then " + title.Offset + " after nine downs, 'more above' shown " + above + ", play says '" + play.text + "'");
        yield return Shot("list");
        foreach (var w in extras) WorldLibrary.Delete(w.id);
        typeof(TitleMenu).GetMethod("Refresh", flags).Invoke(title, null);
        yield return new WaitForSecondsRealtime(4f);
        Mark("MORE: overcast " + overcastThen.ToString("F3") + " then " + TimeOfDay.Instance.Overcast.ToString("F3") + " seven seconds on | gulls " + BirdSong.Gulls);

        float frames = 0f, seconds = 0f;
        for (float t = 0f; t < 3f; t += Time.unscaledDeltaTime) { frames++; seconds += Time.unscaledDeltaTime; yield return null; }
        Mark("TITLE frame " + (seconds / frames * 1000f).ToString("F1") + " ms");

        var image = GameObject.Find("Logo picture");
        var camGo = GameObject.Find("Title logo camera");
        if (camGo != null)
        {
            var tex = camGo.GetComponent<Camera>().targetTexture;
            RenderTexture.active = tex;
            var t2 = new Texture2D(tex.width, tex.height, TextureFormat.RGBA32, false);
            t2.ReadPixels(new Rect(0, 0, tex.width, tex.height), 0, 0); t2.Apply();
            RenderTexture.active = null;
            var px = t2.GetPixels32(); int opaque = 0, left = tex.width, right = 0;
            for (int i = 0; i < px.Length; i++) { if (px[i].a <= 128) continue; opaque++; int x = i % tex.width; left = Mathf.Min(left, x); right = Mathf.Max(right, x); }
            Mark("LOGO picture opaque " + opaque + " of " + px.Length + " | spans x " + left + ".." + right + " of " + tex.width + " | image under " + (image != null && image.transform.parent != null ? image.transform.parent.name : "nothing"));
        }

        yield return Shot("a");
        var cardGo = GameObject.Find("Card");
        if (cardGo != null)
        {
            cardGo.SetActive(false);
            foreach (var when in new[] { "afternoon", "morning", "dusk", "night" })
            {
                title.Look(when);
                yield return new WaitForSecondsRealtime(2f);
                yield return Shot(when);
            }
            title.Look("afternoon");
            cardGo.SetActive(true);
        }
        yield return new WaitForSecondsRealtime(3f);
        yield return Shot("c");

        // a seed previewed from the new page: the backdrop comes back as that seed, the page as it was
        typeof(TitleMenu).GetMethod("ShowNew", flags).Invoke(title, null);
        ((TMPro.TMP_InputField)typeof(TitleMenu).GetField("seedField", flags).GetValue(title)).text = "5";
        ((TMPro.TMP_InputField)typeof(TitleMenu).GetField("nameField", flags).GetValue(title)).text = "Probe title";
        typeof(TitleMenu).GetMethod("Preview", flags).Invoke(title, null);
        yield return new WaitForSecondsRealtime(15f);
        title = FindFirstObjectByType<TitleMenu>();
        var newPage = (GameObject)typeof(TitleMenu).GetField("newPage", flags).GetValue(title);
        var seedField = (TMPro.TMP_InputField)typeof(TitleMenu).GetField("seedField", flags).GetValue(title);
        Mark("PREVIEW: backdrop seed " + FindFirstObjectByType<ChunkManager>().WorldSeed + " | new page up " + newPage.activeSelf + " | seed field '" + seedField.text + "' | " + State());
        yield return Shot("preview");

        // in, and back out, with a picture taken in the world
        typeof(TitleMenu).GetMethod("ShowWorlds", flags).Invoke(title, null);
        var save = WorldLibrary.Create("Probe title", 5, true, true, 0.5f, 20f, true, true);
        typeof(TitleMenu).GetMethod("Enter", flags).Invoke(title, new object[] { save });
        yield return new WaitForSecondsRealtime(0.4f);
        Mark("LEAVING: " + Early(title));
        yield return new WaitForSecondsRealtime(4f);
        var arrival = FindFirstObjectByType<Arrival>();
        if (arrival != null) typeof(Arrival).GetMethod("Close", flags).Invoke(arrival, null);
        yield return new WaitForSecondsRealtime(9f);
        Mark("IN: " + State() + " | pictures " + SaveCoordinator.Pictures + " | file " + File.Exists(WorldLibrary.PicturePath(save.id)));
        yield return Shot("in");
        WorldLibrary.LeaveToMenu();
        yield return new WaitForSecondsRealtime(14f);
        var row = GameObject.Find("World Probe title");
        Mark("BACK: " + State() + " | row picture " + (row != null && row.transform.Find("Picture") != null));
        yield return Shot("back");
        WorldLibrary.Delete(save.id);
        Mark("done");
    }
}
