// A probe: renders a cubemap of the country from a good spot -- by a lake
// with a lighthouse on it, at dawn, weather off, the view at its widest --
// and writes the six faces out for the title to turn slowly behind the menu.
using System.Collections;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        Mark("boot in " + UnityEngine.SceneManagement.SceneManager.GetActiveScene().name);
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);

        // from the title, into whichever world is first
        if (TitleMenu.IsUp)
        {
            var worlds = WorldLibrary.All();
            WorldLibrary.Enter(worlds.Count > 0 ? worlds[0] : WorldLibrary.Create("Panorama", 7777));
            yield return new WaitForSecondsRealtime(6f);
        }

        ChunkManager world = FindFirstObjectByType<ChunkManager>();
        for (int tries = 0; world == null || world.PlayerTransform == null; tries++)
        {
            if (tries > 40) { Mark("gave up"); Mark("done"); yield break; }
            yield return new WaitForSecondsRealtime(1f);
            world = FindFirstObjectByType<ChunkManager>();
        }
        Transform player = world.PlayerTransform;
        var body = player.GetComponent<CharacterController>();
        int seed = world.WorldSeed;
        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;

        // drawn much further than the game ever draws, for the shot alone, so
        // the edge of the drawn world lies beyond the camera's far plane
        typeof(ChunkManager).GetField("viewRadius", flags).SetValue(world, 11);
        typeof(ChunkManager).GetMethod("RefreshVisibleChunks", flags).Invoke(world, new object[] { true });
        if (TimeOfDay.Instance != null) typeof(TimeOfDay).GetField("weather", flags).SetValue(TimeOfDay.Instance, false);
        // fog pulled in short of the edge of the drawn world, so the horizon fades rather than stops
        if (TimeOfDay.Instance != null)
        {
            TimeOfDay.Instance.Paused = true;
            typeof(TimeOfDay).GetField("clearFogStart", flags).SetValue(TimeOfDay.Instance, 50f);
            typeof(TimeOfDay).GetField("clearFogEnd", flags).SetValue(TimeOfDay.Instance, 200f);
        }

        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
        foreach (var rend in player.GetComponentsInChildren<Renderer>()) rend.enabled = false;

        // four views: a structure to stand near, the hour, where to stand from it, the name
        var views = new (int kind, float hour, Vector3 off, float up, string name)[]
        {
            (4, 0.30f, new Vector3(-12f, 0f, 5f), 3.4f, "1-dawn"),
            (0, 0.38f, new Vector3(-20f, 0f, 13f), 2.4f, "2-morning"),
            (8, 0.46f, new Vector3(-17f, 0f, 11f), 2.6f, "3-noon"),
            (12, 0.72f, new Vector3(-13f, 0f, 9f), 3.4f, "4-dusk"),
            (11, 0.965f, new Vector3(-10f, 0f, 7f), 3.2f, "5-night"),
        };

        var home = WorldGrid.WorldToChunk(player.position);
        var rt = new RenderTexture(2048, 2048, 24) { dimension = UnityEngine.Rendering.TextureDimension.Cube };
        rt.Create();
        var face = new Texture2D(2048, 2048, TextureFormat.RGB24, false);

        foreach (var v in views)
        {
            Landmarks.Placement at = default; bool found = false;
            for (int r = 0; r < 90 && !found; r++)
            for (int dx = -r; dx <= r && !found; dx++)
            for (int dz = -r; dz <= r && !found; dz++)
            {
                if (Mathf.Max(Mathf.Abs(dx), Mathf.Abs(dz)) != r) continue;
                var p = Landmarks.In(new Vector2Int(home.x + dx, home.y + dz), seed);
                if (p.Exists && p.Kind == (LandmarkKind)v.kind) { at = p; found = true; }
            }
            Mark(v.name + ": structure " + (LandmarkKind)v.kind + " found " + found);
            if (!found) continue;

            if (TimeOfDay.Instance != null) TimeOfDay.Instance.SetTime(v.hour);
            var turn = Quaternion.Euler(0f, at.Yaw, 0f);
            Vector3 stand = at.Position + turn * v.off;
            int tx = Mathf.RoundToInt(stand.x / WorldGrid.TileSize), tz = Mathf.RoundToInt(stand.z / WorldGrid.TileSize);
            float ground = Mathf.Max(WorldHeight.SurfaceY(tx, tz, seed), WaterSurface.Level);
            body.enabled = false; player.position = new Vector3(stand.x, ground + 0.5f, stand.z); body.enabled = true;
            yield return new WaitForSecondsRealtime(38f);

            foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
            foreach (var beh in FindObjectsByType<MonoBehaviour>(FindObjectsSortMode.None))
                if (beh.GetType().Name == "WorldLabels" || beh.GetType().Name == "Notices" || beh.GetType().Name == "CompassBar") beh.enabled = false;
            foreach (var an in FindObjectsByType<Animal>(FindObjectsSortMode.None)) if (an.Kind != FaunaKind.Deer && an.Kind != FaunaKind.Heron) Destroy(an.gameObject);
            yield return new WaitForSecondsRealtime(1f);

            // The spot itself: within twenty metres of the structure, the tile
            // that suits the view -- sand by the water at dusk, snow at noon,
            // otherwise any dry ground -- with the fewest tall things about it
            // and the structure still in sight.
            {
                int TallNear(Vector3 at, float within)
                {
                    int count = 0;
                    try
                    {
                        var under = FindFirstObjectByType<Undergrowth>();
                        var every = (Flora.Sprout[])typeof(Undergrowth).GetField("every", flags).GetValue(under);
                        var gathered = (Matrix4x4[][])typeof(Undergrowth).GetField("gathered", flags).GetValue(under);
                        for (int i = 0; every != null && gathered != null && i < every.Length && i < gathered.Length; i++)
                        {
                            if (gathered[i] == null) continue;
                            foreach (var m in gathered[i])
                            {
                                if (m.lossyScale.y * every[i].Size < 1.4f) continue;
                                var pos = m.GetPosition(); pos.y = 0f; var flat = at; flat.y = 0f;
                                if (Vector3.Distance(pos, flat) < within) count++;
                            }
                        }
                    }
                    catch { }
                    return count;
                }
                int sx = Mathf.RoundToInt(at.Position.x / WorldGrid.TileSize), sz = Mathf.RoundToInt(at.Position.z / WorldGrid.TileSize);
                var about = Landmarks.All(at.Kind);
                float clear = Mathf.Max(about.Ahead, Mathf.Max(about.Behind, about.Aside)) * WorldGrid.TileSize + 8f;   // past the whole of the structure
                Vector3 bestSpot = stand; float bestSpotScore = float.MaxValue;
                for (int dx = -22; dx <= 22; dx++) for (int dz = -22; dz <= 22; dz++)
                {
                    int tx2 = sx + dx, tz2 = sz + dz;
                    float d = Mathf.Sqrt(dx * dx + dz * dz) * WorldGrid.TileSize;
                    if (d < clear || d > clear + 18f) continue;
                    // flat about it: no terrace steps within four tiles
                    float lo = float.MaxValue, hi = float.MinValue;
                    for (int i = -4; i <= 4; i++) for (int j = -4; j <= 4; j++) { float hh = WorldHeight.SurfaceY(tx2 + i, tz2 + j, seed); lo = Mathf.Min(lo, hh); hi = Mathf.Max(hi, hh); }
                    if (hi - lo > 0.3f) continue;
                    bool wet = WaterSurface.IsUnderwater(tx2, tz2, seed);
                    bool snowy = SnowCover.IsSnowy(tx2, tz2, seed);
                    var here = Regions.CharacterAtTile(tx2, tz2, seed);
                    bool suits = v.name.Contains("dusk") ? (!wet && here == Regions.Character.Water)
                               : v.name.Contains("noon") ? (!wet && snowy)
                               : (!wet && !snowy);
                    if (!suits) continue;
                    var p3 = new Vector3(tx2 * WorldGrid.TileSize, 0f, tz2 * WorldGrid.TileSize);
                    float score = TallNear(p3, 8f) * 10f + TallNear(p3, 14f) + Mathf.Abs(d - clear - 6f) * 0.2f;
                    if (score < bestSpotScore) { bestSpotScore = score; bestSpot = p3; }
                }
                stand = bestSpot;
                tx = Mathf.RoundToInt(stand.x / WorldGrid.TileSize); tz = Mathf.RoundToInt(stand.z / WorldGrid.TileSize);
                ground = Mathf.Max(WorldHeight.SurfaceY(tx, tz, seed), WaterSurface.Level);
                Mark(v.name + ": spot " + stand.ToString("F0") + " score " + bestSpotScore.ToString("F1"));
            }
            cam.transform.position = new Vector3(stand.x, ground + v.up, stand.z);
            cam.transform.rotation = Quaternion.identity;
            cam.farClipPlane = 240f;
            if (TimeOfDay.Instance != null) TimeOfDay.Instance.UnlockFog();
            yield return null;

            // a small render first, to read the sky's colour at the horizon:
            // the fog is then held to it, and the drawn world fades into sky
            var small = new RenderTexture(256, 256, 24) { dimension = UnityEngine.Rendering.TextureDimension.Cube };
            small.Create();
            cam.RenderToCubemap(small, 63);
            var peek = new Texture2D(256, 256, TextureFormat.RGB24, false);
            Color sky = Color.gray;
            float best = -1f;
            foreach (int f in new[] { 0, 1, 4, 5 })
            {
                Graphics.SetRenderTarget(small, 0, (CubemapFace)f);
                peek.ReadPixels(new Rect(0, 0, 256, 256), 0, 0); peek.Apply();
                foreach (int row in new[] { 128 - 14, 128 + 14 })
                {
                    Color sum = Color.black; int n = 0;
                    for (int x = 16; x < 240; x += 4) { sum += peek.GetPixel(x, row); n++; }
                    var c = sum / n;
                    float skyness = c.grayscale;                    // the sky is the brighter band, at any hour
                    if (skyness > best) { best = skyness; sky = c; }
                }
            }
            Graphics.SetRenderTarget(null);
            small.Release();
            if (TimeOfDay.Instance != null) TimeOfDay.Instance.LockFog(sky);
            Mark(v.name + ": horizon sky " + ColorUtility.ToHtmlStringRGB(sky));
            yield return null; yield return null;

            // Six faces, each rendered with the camera actually turned that way
            // and a frame allowed to pass first: the tiles, the plants and the
            // animals are only submitted for what the camera can see, so a
            // cubemap taken all at once had a world in one face and void in
            // the other five.
            var face2d = new RenderTexture(2048, 2048, 24);
            face2d.Create();
            var faceTex = new Texture2D(2048, 2048, TextureFormat.RGB24, false);
            cam.fieldOfView = 90f;
            cam.aspect = 1f;
            var looks = new Quaternion[]
            {
                Quaternion.Euler(0f, 90f, 0f),     // +X
                Quaternion.Euler(0f, -90f, 0f),    // -X
                Quaternion.Euler(-90f, 0f, 0f),    // +Y, up
                Quaternion.Euler(90f, 0f, 0f),     // -Y, down
                Quaternion.Euler(0f, 0f, 0f),      // +Z
                Quaternion.Euler(0f, 180f, 0f),    // -Z
            };
            bool ok = true;
            for (int f = 0; f < 6; f++)
            {
                cam.transform.rotation = looks[f];
                yield return null; yield return null;          // the world catches up with the turn
                cam.targetTexture = face2d;
                cam.Render();
                RenderTexture.active = face2d;
                faceTex.ReadPixels(new Rect(0, 0, 2048, 2048), 0, 0);
                faceTex.Apply();
                RenderTexture.active = null;
                cam.targetTexture = null;
                File.WriteAllBytes(Path.Combine(Application.persistentDataPath, "pano-" + v.name + "-" + f + ".png"), faceTex.EncodeToPNG());
            }
            cam.ResetAspect();
            face2d.Release();
            Mark(v.name + ": rendered " + ok);
        }
        Mark("done");
    }
}
