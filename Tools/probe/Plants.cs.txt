// A probe: what the undergrowth plants now. In a fungal country and a lowland, every kind with instances
// in reach, and whether any of them is still the pack's; photographs of both.
using System.Collections;
using System.Collections.Generic;
using System.IO;
using UnityEngine;

public class _Probe : MonoBehaviour
{
    private const string Stages = "__STAGES__";

    [RuntimeInitializeOnLoadMethod(RuntimeInitializeLoadType.AfterSceneLoad)]
    private static void Boot()
    {
        Application.runInBackground = true;
        if (FindFirstObjectByType<_Probe>() != null) return;
        var go = new GameObject("_Probe");
        DontDestroyOnLoad(go);
        go.AddComponent<_Probe>();
    }

    private static void Mark(string what)
    {
        try { File.AppendAllText(Stages, System.DateTime.Now.ToString("HH:mm:ss ") + what + "\n"); } catch { }
    }

    private static readonly string[] Ours = { "Mushroom", "Toadstool", "Boulder", "Desert Stone", "Dead Tree", "Oak", "Beech", "Birch", "Sapling", "Pine", "Snow ", "Saguaro", "Barrel", "Prickly", "Palm", "Reeds" };

    private IEnumerator Shot(string name, Camera cam, Vector3 at, Vector3 lookAt)
    {
        for (int i = 0; i < 2; i++) { cam.transform.position = at; cam.transform.LookAt(lookAt); yield return null; }
        yield return new WaitForEndOfFrame();
        ScreenCapture.CaptureScreenshot(Path.Combine(Application.persistentDataPath, "probe-pl-" + name + ".png"));
        yield return new WaitForSecondsRealtime(0.3f);
    }

    private string Planted(Undergrowth under, System.Reflection.BindingFlags flags)
    {
        var every = (Flora.Sprout[])typeof(Undergrowth).GetField("every", flags).GetValue(under);
        var gathered = (Matrix4x4[][])typeof(Undergrowth).GetField("gathered", flags).GetValue(under);
        var said = new System.Text.StringBuilder(); int pack = 0, all = 0;
        for (int k = 0; k < every.Length && k < gathered.Length; k++)
        {
            int n = gathered[k] != null ? gathered[k].Length : 0;
            if (n == 0) continue;
            all += n;
            string name = every[k].Name ?? "";
            bool ours = false; foreach (var o in Ours) if (name.StartsWith(o)) ours = true;
            if (!ours) pack += n;
            said.Append(" " + name + ":" + n);
        }
        return "planted " + all + ", the pack's " + pack + " |" + said;
    }

    private IEnumerator Start()
    {
        yield return new WaitForSecondsRealtime(3f);
        var save = WorldLibrary.Create("Probe plants", 24, true, true, 0.5f, 20f, true, true);
        WorldLibrary.Enter(save);
        yield return new WaitForSecondsRealtime(9f);
        var flags = System.Reflection.BindingFlags.NonPublic | System.Reflection.BindingFlags.Instance;
        var arrival = FindFirstObjectByType<Arrival>();
        if (arrival != null) typeof(Arrival).GetMethod("Close", flags).Invoke(arrival, null);
        var tools = FindFirstObjectByType<DevTools>();
        var goTo = typeof(DevTools).GetMethod("GoTo", flags, null, new[] { typeof(Regions.Character) }, null);
        var world = FindFirstObjectByType<ChunkManager>(); var player = world.PlayerTransform; var under = FindFirstObjectByType<Undergrowth>();
        var cam = Camera.main;
        foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
        foreach (var c in FindObjectsByType<Canvas>(FindObjectsSortMode.None)) c.enabled = false;
        foreach (var where in new[] { Regions.Character.Fungal, Regions.Character.Lowland, Regions.Character.Desert, Regions.Character.Dead })
        {
            foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = true;
            goTo.Invoke(tools, new object[] { where });
            yield return new WaitForSecondsRealtime(8f);
            foreach (var beh in cam.GetComponents<Behaviour>()) if (beh.GetType().Name == "SimpleFollowCamera") beh.enabled = false;
            Mark("PLANTS " + where + " at " + player.position.ToString("F0") + ": " + Planted(under, flags));
            Vector3 fwd = player.forward;
            yield return Shot(where.ToString().ToLowerInvariant(), cam, player.position + Vector3.up * 1.7f - fwd * 2.5f, player.position + fwd * 8f + Vector3.up * 1.2f);
        }
        Mark("done");
        WorldLibrary.Delete(save.id);
    }
}
